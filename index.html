function changeLanguage(lang) { const words = T[lang] || T.uz; document.documentElement.lang = lang; document.querySelectorAll("[data-t]").forEach(function(el) { const key = el.getAttribute("data-t"); if (words[key]) { el.textContent = words[key]; } }); try { localStorage.setItem("bukharaTourLanguage", lang); } catch (e) {} } const select = document.getElementById("language"); select.addEventListener("change", function() { changeLanguage(select.value); }); let saved = "uz"; try { saved = localStorage.getItem("bukharaTourLanguage") || "uz"; } catch (e) {} if (!T[saved]) { saved = "uz"; } select.value = saved; changeLanguage(saved); function contact() { alert(T[select.value].bookingD); }
